\section{半正定矩阵与特征值}

半正定矩阵和正定矩阵的一个重要性质是，其特征值分别大于等于或大于零！

\begin{BoxTheorem}[半正定矩阵和特征值]
    设$\vb{A}\in\Symm^n$，记$\lambda_1,\cdots,\lambda_n$是$\vb{A}$的特征值，则
    \begin{itemize}
        \item $\vb{A}\succ\vb{0}$，当且仅当对于$i=1,\cdots,n$有$\lambda_i>0$。
        \item $\vb{A}\succeq\vb{0}$，当且仅当对于$i=1,\cdots,n$有$\lambda_i\geq 0$。
        \item $\vb{A}\not\succeq\vb{0}$，当且仅当存在$\lambda_i>0$和$\lambda_i<0$。
    \end{itemize}
\end{BoxTheorem}

\begin{Proof}[\cref{thm:半正定矩阵和特征值}]
    根据\cref{thm:厄米矩阵的特征分解}，半正定矩阵作为对称矩阵一定存在特征分解
    \begin{Equation}[半正定矩阵和特征值1]
        \vb{A}=\vb{V}\vb*{\Lambda}\vb{V}^{-1}=\vb{V}\vb*{\Lambda}\vb{V}^{T}
    \end{Equation}

    因此，$\vb{A}\succeq\vb{0}$就等价于对于任意的$\vb{x}\in\R^n$
    \begin{Equation}[]
        \vb{x}^T\vb{V}\vb*{\Lambda}\vb{V}^T\vb{x}\geq 0
    \end{Equation}

    不妨令$\vb{z}=\vb{V}^T\vb{x}$，注意到$\vb{z}\in\rangespace(\vb{V}^T)$即$\vb{z}\in\R$
    \begin{Equation}[]
        \vb{z}^T\vb*{\Lambda}\vb{z}\geq 0
    \end{Equation}

    按列向量展开
    \begin{Equation}[]
        \Sum[i=1][n]\lambda_i\abs{z_i}^2\geq 0
    \end{Equation}

    由于需要对于任意的$\vb{z}\in\R^n$成立，因此必须有$\lambda_i\geq 0$对于$i=1,\cdots,n$成立。
\end{Proof}
